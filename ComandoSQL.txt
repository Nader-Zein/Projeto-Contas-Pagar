CREATE TABLE CondicaoPagamento (
    IdCondPgto INT PRIMARY KEY AUTO_INCREMENT,
    Descricao VARCHAR(40) NOT NULL,
    QuantidadeParcelas INT NOT NULL,
    Juros DECIMAL(5,2) NOT NULL DEFAULT 0,
    Multa DECIMAL(5,2) NOT NULL DEFAULT 0,
    Desconto DECIMAL(5,2) NOT NULL DEFAULT 0,
    Status BOOLEAN NOT NULL,
    DataCriacao DATETIME NOT NULL,
    DataEdicao DATETIME NULL
);


CREATE TABLE FormaPagamento (
    IdFormaPgto INT PRIMARY KEY AUTO_INCREMENT,
    Descricao VARCHAR(40) NOT NULL,
    Status BOOLEAN NOT NULL,
    DataCriacao DATETIME NOT NULL,
    DataEdicao DATETIME NULL
);

CREATE TABLE ParcelaCondicaoPagamento (
    IdCondPgto INT NOT NULL,
    NumeroParcela INT NOT NULL,
    IdFormaPgto INT NOT NULL,
    ValorPercentual DECIMAL(10,2) NOT NULL,
    DiasAposVenda INT NOT NULL,
    PRIMARY KEY (IdCondPgto, NumeroParcela, IdFormaPgto),
    FOREIGN KEY (IdCondPgto) REFERENCES CondicaoPagamento(IdCondPgto) ON DELETE CASCADE,
    FOREIGN KEY (IdFormaPgto) REFERENCES FormaPagamento(IdFormaPgto)
);

CREATE TABLE Produto (
    IdProduto INT PRIMARY KEY AUTO_INCREMENT,
    Descricao VARCHAR(40) NOT NULL,

    Referencia VARCHAR(10) NOT NULL,

    CategoriaId INT NOT NULL,
    ValorVenda DECIMAL(10,2) NOT NULL,
    CustoMedio DECIMAL(10,2) NOT NULL DEFAULT 0, 
    CustoUltimaCompraDECIMAL(10,2) NOT NULL DEFAULT 0, 
    Quantidade INT NOT NULL,
    QuantidadeMinima INT NOT NULL,

    Status BOOLEAN NOT NULL,

CREATE TABLE ProdutoFornecedor (
    IdProduto INT NOT NULL,
    IdFornecedor INT NOT NULL,
    CustoUltimaCompra DECIMAL(10,2) NOT NULL DEFAULT 0, rateio
    DataUltimaCompra DATETIME NULL,
    Observacao VARCHAR(40) NULL,
    PRIMARY KEY (IdProduto, IdFornecedor),
    FOREIGN KEY (IdProduto) REFERENCES Produto(IdProduto) ON DELETE CASCADE,
    FOREIGN KEY (IdFornecedor) REFERENCES Fornecedor(IdFornecedor) ON DELETE CASCADE
);



CREATE TABLE Compra (
    
    Modelo VARCHAR(10) NOT NULL,
    Serie VARCHAR(10) NOT NULL,
    NumeroNota INT NOT NULL,
    FornecedorId INT NOT NULL,
    
   
    DataEmissao DATETIME NOT NULL,
    DataChegada DATETIME NOT NULL,
    CondicaoPagamentoId INT NOT NULL,
    Frete DECIMAL(10,2) DEFAULT 0,
    Seguro DECIMAL(10,2) DEFAULT 0,
    Despesas DECIMAL(10,2) DEFAULT 0,
    Status BOOLEAN NOT NULL,
    Observacao VARCHAR(40) NULL,
    -- Campos de Auditoria
    DataCriacao DATETIME NOT NULL,
    

    
    PRIMARY KEY (Modelo, Serie, NumeroNota, FornecedorId),
    FOREIGN KEY (FornecedorId) REFERENCES Fornecedor(IdFornecedor),
    FOREIGN KEY (CondicaoPagamentoId) REFERENCES CondicaoPagamento(IdCondPgto)
);



CREATE TABLE ItemCompra (
    
    CompraModelo VARCHAR(10) NOT NULL,
    CompraSerie VARCHAR(10) NOT NULL,
    CompraNumeroNota INT NOT NULL,
    CompraFornecedorId INT NOT NULL,

   
    ProdutoId INT NOT NULL,

    
    Quantidade INT NOT NULL,
    ValorUnitario DECIMAL(10,2) NOT NULL,
    CustoAdicionalUnitario DECIMAL(18,2) NOT NULL DEFAULT 0,
    CustoUnitarioReal     DECIMAL(18,2) NOT NULL DEFAULT 0, 	
    
    PRIMARY KEY (CompraModelo, CompraSerie, CompraNumeroNota, CompraFornecedorId, ProdutoId),

    
    FOREIGN KEY (CompraModelo, CompraSerie, CompraNumeroNota, CompraFornecedorId) 
        REFERENCES Compra(Modelo, Serie, NumeroNota, FornecedorId) 
        ON DELETE CASCADE, -- Se a compra for deletada, os itens são deletados juntos.
    FOREIGN KEY (ProdutoId) REFERENCES Produto(IdProduto)
);



CREATE TABLE Conta_a_Pagar (
    Modelo VARCHAR(10) NOT NULL,        -- Parte da FK para Compra e PK. Ajustado tamanho para 10.
    Serie VARCHAR(10) NOT NULL,         -- Parte da FK para Compra e PK. Ajustado tamanho para 10.
    NumeroNota INT NOT NULL,            -- Parte da FK para Compra e PK.
    FornecedorId INT NOT NULL,          -- Parte da FK para Compra e PK. RENOMEADO de CodFornecedor.
    NumeroParcela INT NOT NULL,         -- Identificador da parcela dentro da nota (e PK).

    Status BOOLEAN NOT NULL, 
    Situacao  VARCHAR(25) NULL,
    ValorParcela DECIMAL(10,2) NOT NULL, -- Valor calculado da parcela.
    DataVencimento DATE NOT NULL,         -- Data de vencimento calculada.
    DataEmissao DATE NOT NULL,            -- Data de Emissão da Nota original (para referência).

   
    DataPagamento DATE NULL,              -- Data em que a parcela foi paga (NULL se não paga).
    Juros DECIMAL(10,2) NULL DEFAULT 0.00,  -- Juros aplicados no pagamento.
    Multa DECIMAL(10,2) NULL DEFAULT 0.00,  -- Multa aplicada no pagamento.
    Desconto DECIMAL(10,2) NULL DEFAULT 0.00, -- Desconto concedido no pagamento.
    ValorPago DECIMAL(10,2) NULL,         -- Valor total efetivamente pago.
    IdFormaPgto INT NULL,                 -- Forma como foi pago (NULL se não pago).

    
    PRIMARY KEY (Modelo, Serie, NumeroNota, FornecedorId, NumeroParcela),

    
    FOREIGN KEY (Modelo, Serie, NumeroNota, FornecedorId) -- Colunas correspondentes na Conta_a_Pagar
        REFERENCES Compra(Modelo, Serie, NumeroNota, FornecedorId) -- Referencia a tabela Compra e suas colunas PK
        ON DELETE CASCADE, -- Se a compra for excluída, as contas a pagar são excluídas.
    FOREIGN KEY (IdFormaPgto) REFERENCES FormaPagamento(IdFormaPgto)
);
