CREATE TABLE CondicaoPagamento (
    IdCondPgto INT PRIMARY KEY AUTO_INCREMENT,
    Descricao VARCHAR(40) NOT NULL,
    QuantidadeParcelas INT NOT NULL,
    Juros DECIMAL(5,2) NOT NULL DEFAULT 0,
    Multa DECIMAL(5,2) NOT NULL DEFAULT 0,
    Desconto DECIMAL(5,2) NOT NULL DEFAULT 0,
    Status BOOLEAN NOT NULL,
    DataCriacao DATETIME NOT NULL,
    DataEdicao DATETIME NULL
);


CREATE TABLE FormaPagamento (
    IdFormaPgto INT PRIMARY KEY AUTO_INCREMENT,
    Descricao VARCHAR(40) NOT NULL,
    Status BOOLEAN NOT NULL,
    DataCriacao DATETIME NOT NULL,
    DataEdicao DATETIME NULL
);

CREATE TABLE ParcelaCondicaoPagamento (
    IdCondPgto INT NOT NULL,
    NumeroParcela INT NOT NULL,
    IdFormaPgto INT NOT NULL,
    ValorPercentual DECIMAL(10,2) NOT NULL,
    DiasAposVenda INT NOT NULL,
    PRIMARY KEY (IdCondPgto, NumeroParcela, IdFormaPgto),
    FOREIGN KEY (IdCondPgto) REFERENCES CondicaoPagamento(IdCondPgto) ON DELETE CASCADE,
    FOREIGN KEY (IdFormaPgto) REFERENCES FormaPagamento(IdFormaPgto)
);

   media ponderada PrecoMedioCusto DECIMAL(10,2) NULL, = valor de compra na tela = valor unitario mais rateio, este valor soma e devide por 2 com o anterior
   rateio PrecoUltimaCompra DECIMAL(10,2) NULL,

CREATE TABLE Compra (
    -- Chave Primária Composta, que identifica unicamente uma Nota Fiscal de um Fornecedor
    Modelo VARCHAR(10) NOT NULL,
    Serie VARCHAR(10) NOT NULL,
    NumeroNota INT NOT NULL,
    FornecedorId INT NOT NULL,
    
    -- Dados da Compra
    DataEmissao DATETIME NOT NULL,
    DataChegada DATETIME NOT NULL,
    CondicaoPagamentoId INT NOT NULL,
    Frete DECIMAL(10,2) DEFAULT 0,
    Seguro DECIMAL(10,2) DEFAULT 0,
    Despesas DECIMAL(10,2) DEFAULT 0,
    Status BOOLEAN NOT NULL,
    
    -- Campos de Auditoria
    DataCriacao DATETIME NOT NULL,
    

    -- Definição da Chave Primária e Chaves Estrangeiras
    PRIMARY KEY (Modelo, Serie, NumeroNota, FornecedorId),
    FOREIGN KEY (FornecedorId) REFERENCES Fornecedor(IdFornecedor),
    FOREIGN KEY (CondicaoPagamentoId) REFERENCES CondicaoPagamento(IdCondPgto)
);



CREATE TABLE ItemCompra (
    -- Chave Estrangeira Composta que referencia a tabela Compra
    CompraModelo VARCHAR(10) NOT NULL,
    CompraSerie VARCHAR(10) NOT NULL,
    CompraNumeroNota INT NOT NULL,
    CompraFornecedorId INT NOT NULL,

    -- Parte da Chave Primária que referencia o Produto
    ProdutoId INT NOT NULL,

    -- Dados do Item
    Quantidade INT NOT NULL,
    ValorUnitario DECIMAL(10,2) NOT NULL,

    -- Definição da Chave Primária (5 colunas para garantir que um produto não se repita na mesma nota)
    PRIMARY KEY (CompraModelo, CompraSerie, CompraNumeroNota, CompraFornecedorId, ProdutoId),

    -- Definição das Chaves Estrangeiras
    FOREIGN KEY (CompraModelo, CompraSerie, CompraNumeroNota, CompraFornecedorId) 
        REFERENCES Compra(Modelo, Serie, NumeroNota, FornecedorId) 
        ON DELETE CASCADE, -- Se a compra for deletada, os itens são deletados juntos.
    FOREIGN KEY (ProdutoId) REFERENCES Produto(IdProduto)
);
